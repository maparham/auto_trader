**Goal:** Add a causal `SWEEP` chart indicator that flags liquidity-sweep (stop hunt) candles. It is drawn on the chart and works as an operand in backtest rules, screens and alerts.

**Architecture:** A per-bar state machine (pivot and previous-day levels, sweep detection, score, confirmation) lives in a node-safe TS leaf (`sweepCompute.ts`) and is mirrored operation-for-operation in Python (`indicators/sweep.py`). A golden fixture pins the two to each other. The chart template (`sweep.ts`) only draws. Registration follows the SPIKE pattern: backend series registry plus alert spec, and frontend template, meta, expr instance wiring and the timezone stamp.

**Tech Stack:** Python 3 (pytest), TypeScript, vitest, klinecharts.

**Spec:** `docs/superpowers/specs/2026-10-23-sweep-indicator-design.md`

## Review Focus

1. **Outside bar sweeping both sides with equal scores.** Expect neither side to flag, and both levels to die. Pinned in Task 1 and Task 3 (`test_outside_bar_tie_flags_neither`).
2. **A level price broke out of earlier (close beyond it).** Expect a later close back through it NOT to count as a sweep. Pinned in Task 1 and Task 3 (`test_breakout_kills_level`).
3. **Weekly or daily+ bars with `usePrevDay` on.** Expect no previous-day levels (spacing above one day blanks the source). Pinned in Task 1 (`test_prev_day_blank_on_weekly_bars`).
4. **The chart zone moves the day boundary.** Expect previous-day levels to follow the pane tz on both sides. Pinned in Task 1 (`test_prev_day_follows_tz`) and in the golden's New York case (Task 3).
5. **A new flag while an opposite-side sweep is pending.** Expect the old sweep never to confirm. Pinned in Task 1 and Task 3 (`test_new_flag_replaces_pending`).

```python
"""SWEEP instances (`SWEEP.bearSweep` / `.bullSweep` / `.bearConfirmed` /
`.bullConfirmed` / `.score` / `.sweptLevel` / `.barsSince`): a causal
liquidity-sweep (stop hunt) detector, mirrored operation-for-operation in
frontend lib/indicators/sweepCompute.ts per the parity contract in
indicators/core.py. Chart-timeframe only (no MTF pin).

Levels are strict fractal pivots, born `pivot_right` bars after the pivot bar
(when they confirm), plus the previous day's high and low in the pane tz,
born on the first bar of the new day. A level is live while
`i - born <= level_max_age`. Per bar: births, expiry, crossings, the confirm
check of the pending sweep, then this bar's flag.

A high level dies on the first bar whose high trades above it, whatever the
close; every level crossed on that bar dies and the highest one is the bear
candidate. It is a sweep when the bar closes back below it, and it flags when
its score reaches `min_score`. Lows mirror this. A bar that flags both ways
takes the higher score; an exact tie flags neither.

A flagged sweep becomes pending and confirms `confirm_bars` later if no close
went back through its level; a close through drops it, and any new flag
replaces it."""

from __future__ import annotations

import math
from collections.abc import Sequence
from dataclasses import dataclass
from datetime import datetime
from zoneinfo import ZoneInfo

from auto_trader.core.models import Candle
from auto_trader.indicators.core import atr_series
from auto_trader.indicators.prev_hl import _zone, estimate_bar_ms

# Pane order. `bearSweep` first: the chart click-to-insert token emits
# outputs[0], an event that is usable as a rule at once.
SWEEP_OUTPUTS: tuple[str, ...] = (
    "bearSweep", "bullSweep", "bearConfirmed", "bullConfirmed", "score", "sweptLevel", "barsSince",
)

_OUTPUT_INDEX = {name: i for i, name in enumerate(SWEEP_OUTPUTS)}
_DAY_MS = 86_400_000


@dataclass(frozen=True, slots=True)
class SweepConfig:
    pivot_left: int = 5
    pivot_right: int = 5
    level_max_age: int = 150
    min_wick_pct: float = 50.0  # swept-side wick, % of range, for full wick credit
    atr_len: int = 14
    min_range_atr: float = 1.5  # bar range in ATRs for full range credit
    vol_len: int = 20
    vol_z: float = 2.0
    min_score: float = 70.0
    confirm_bars: int = 3
    use_prev_day: bool = True
    use_volume: bool = False
    tz: str = "UTC"


_DEFAULT = SweepConfig()


def parse_sweep_config(calc_params: object, extend_data: object) -> SweepConfig:
    """Mirrors frontend sweepOutputs.parseSweepConfig: lengths fall back on
    anything non-finite or 0 (then floor, at least 1), positive reals on
    anything non-finite or <= 0. calcParams order: [pivotLeft, pivotRight,
    levelMaxAge, minWickPct, atrLen, minRangeAtr, volLen, volZ, minScore,
    confirmBars]. extendData: usePrevDay (absent means on), useVolume (only
    `true` turns it on), tz (PREV_HL's _zone rules)."""
    p = calc_params if isinstance(calc_params, (list, tuple)) else []
    ext = extend_data if isinstance(extend_data, dict) else {}

    def num_at(i: int) -> float:
        try:
            return float(p[i])
        except (IndexError, TypeError, ValueError):
            return float("nan")

    def len_at(i: int, default: int) -> int:
        v = num_at(i)
        if not math.isfinite(v) or v == 0:
            v = default
        return max(1, math.floor(v))

    def pos_at(i: int, default: float) -> float:
        v = num_at(i)
        return v if math.isfinite(v) and v > 0 else default

    d = _DEFAULT
    return SweepConfig(
        pivot_left=len_at(0, d.pivot_left),
        pivot_right=len_at(1, d.pivot_right),
        level_max_age=len_at(2, d.level_max_age),
        min_wick_pct=pos_at(3, d.min_wick_pct),
        atr_len=len_at(4, d.atr_len),
        min_range_atr=pos_at(5, d.min_range_atr),
        vol_len=len_at(6, d.vol_len),
        vol_z=pos_at(7, d.vol_z),
        min_score=pos_at(8, d.min_score),
        confirm_bars=len_at(9, d.confirm_bars),
        use_prev_day=ext.get("usePrevDay") is not False,
        use_volume=ext.get("useVolume") is True,
        tz=_zone(ext.get("tz")),
    )


def _clamp01(x: float) -> float:
    return 0.0 if x < 0 else 1.0 if x > 1 else x


def sweep_score(
    cfg: SweepConfig, bear: bool, o: float, h: float, l: float, c: float,
    atr_prev: float | None, vol_z: float | None,
) -> float:
    """Weighted mean of the clamped components, times 100. Range 0 scores 0;
    a missing ATR or volume reading gives that component 0."""
    rng = h - l
    if rng <= 0:
        return 0.0
    if bear:
        wick = (h - max(o, c)) / rng
        close_d = (h - c) / rng
    else:
        wick = (min(o, c) - l) / rng
        close_d = (c - l) / rng
    w = _clamp01(wick / (cfg.min_wick_pct / 100.0))
    cl = _clamp01(close_d / (2.0 / 3.0))
    r = _clamp01(rng / (cfg.min_range_atr * atr_prev)) if atr_prev is not None and atr_prev > 0 else 0.0
    if cfg.use_volume:
        v = _clamp01(vol_z / cfg.vol_z) if vol_z is not None else 0.0
        return (w * 0.3 + cl * 0.15 + r * 0.3 + v * 0.25) * 100.0
    return (w * 0.4 + cl * 0.2 + r * 0.4) * 100.0


def _vol_z(vols: Sequence[float], i: int, n: int) -> float | None:
    """z of bar i's volume against the n bars before it (population stdev)."""
    if i < n:
        return None
    s = 0.0
    for j in range(i - n, i):
        s += vols[j]
    mean = s / n
    ss = 0.0
    for j in range(i - n, i):
        d = vols[j] - mean
        ss += d * d
    sd = math.sqrt(ss / n)
    if sd == 0:
        return None
    return (vols[i] - mean) / sd


def _is_pivot(vals: Sequence[float], p: int, left: int, right: int, want_high: bool) -> bool:
    """Strict fractal: every neighbour in [p-left, p+right] strictly beyond."""
    v = vals[p]
    for j in range(p - left, p + right + 1):
        if j == p:
            continue
        w = vals[j]
        if want_high:
            if w >= v:
                return False
        elif w <= v:
            return False
    return True


@dataclass(slots=True)
class _Level:
    price: float
    born: int
    start: int


def _compute_points(cfg: SweepConfig, candles: Sequence[Candle]) -> list[tuple[float | None, ...]]:
    n = len(candles)
    opens = [c.open for c in candles]
    highs = [c.high for c in candles]
    lows = [c.low for c in candles]
    closes = [c.close for c in candles]
    vols = [c.volume for c in candles]
    ts = [int(c.time.timestamp() * 1000) for c in candles]
    atr = atr_series(candles, cfg.atr_len)
    day_on = cfg.use_prev_day and estimate_bar_ms(ts) <= _DAY_MS
    zone = ZoneInfo(cfg.tz) if day_on else None

    hi_levels: list[_Level] = []
    lo_levels: list[_Level] = []
    day_key = None
    day_hi = -math.inf
    day_lo = math.inf
    pend_dir = 0
    pend_level = 0.0
    pend_bar = 0
    last_score: float | None = None
    last_level: float | None = None
    last_bar = -1

    out: list[tuple[float | None, ...]] = []
    for i in range(n):
        # Births: the pivot at p confirms on this bar.
        p = i - cfg.pivot_right
        if p - cfg.pivot_left >= 0:
            if _is_pivot(highs, p, cfg.pivot_left, cfg.pivot_right, True):
                hi_levels.append(_Level(highs[p], i, p))
            if _is_pivot(lows, p, cfg.pivot_left, cfg.pivot_right, False):
                lo_levels.append(_Level(lows[p], i, p))
        if zone is not None:
            key = datetime.fromtimestamp(ts[i] / 1000, zone).date()
            if day_key is None:
                day_key = key
            elif key != day_key:
                hi_levels.append(_Level(day_hi, i, i))
                lo_levels.append(_Level(day_lo, i, i))
                day_key = key
                day_hi = -math.inf
                day_lo = math.inf

        # Expiry.
        hi_levels = [lv for lv in hi_levels if i - lv.born <= cfg.level_max_age]
        lo_levels = [lv for lv in lo_levels if i - lv.born <= cfg.level_max_age]

        # Crossings: every crossed level dies; the most extreme is the candidate.
        bear_lv: _Level | None = None
        kept: list[_Level] = []
        for lv in hi_levels:
            if highs[i] > lv.price:
                if bear_lv is None or lv.price > bear_lv.price:
                    bear_lv = lv
            else:
                kept.append(lv)
        hi_levels = kept
        bull_lv: _Level | None = None
        kept = []
        for lv in lo_levels:
            if lows[i] < lv.price:
                if bull_lv is None or lv.price < bull_lv.price:
                    bull_lv = lv
            else:
                kept.append(lv)
        lo_levels = kept

        atr_prev = atr[i - 1] if i >= 1 else None
        vz = _vol_z(vols, i, cfg.vol_len) if cfg.use_volume and (bear_lv or bull_lv) else None
        bear_s: float | None = None
        bull_s: float | None = None
        if bear_lv is not None and closes[i] < bear_lv.price:
            s = sweep_score(cfg, True, opens[i], highs[i], lows[i], closes[i], atr_prev, vz)
            if s >= cfg.min_score:
                bear_s = s
        if bull_lv is not None and closes[i] > bull_lv.price:
            s = sweep_score(cfg, False, opens[i], highs[i], lows[i], closes[i], atr_prev, vz)
            if s >= cfg.min_score:
                bull_s = s
        flag = 0
        if bear_s is not None and bull_s is not None:
            flag = -1 if bear_s > bull_s else 1 if bull_s > bear_s else 0
        elif bear_s is not None:
            flag = -1
        elif bull_s is not None:
            flag = 1

        # Confirm check of the sweep pending from an earlier bar.
        bear_conf = 0.0
        bull_conf = 0.0
        if pend_dir != 0:
            if pend_dir < 0 and closes[i] > pend_level:
                pend_dir = 0
            elif pend_dir > 0 and closes[i] < pend_level:
                pend_dir = 0
            elif i - pend_bar >= cfg.confirm_bars:
                if pend_dir < 0:
                    bear_conf = 1.0
                else:
                    bull_conf = 1.0
                pend_dir = 0

        # This bar's flag: it replaces any pending sweep.
        if flag != 0:
            lv = bear_lv if flag < 0 else bull_lv
            assert lv is not None
            pend_dir = flag
            pend_level = lv.price
            pend_bar = i
            last_score = bear_s if flag < 0 else bull_s
            last_level = lv.price
            last_bar = i

        if zone is not None:
            day_hi = max(day_hi, highs[i])
            day_lo = min(day_lo, lows[i])

        out.append((
            1.0 if flag < 0 else 0.0,
            1.0 if flag > 0 else 0.0,
            bear_conf,
            bull_conf,
            last_score,
            last_level,
            float(i - last_bar) if last_bar >= 0 else None,
        ))
    return out


def sweep_outputs(cfg: SweepConfig) -> tuple[str, ...]:
    return SWEEP_OUTPUTS


def sweep_series(
    cfg: SweepConfig, output: str, candles: Sequence[Candle], bar_hours: float
) -> list[float | None]:
    idx = _OUTPUT_INDEX.get(output, 0)
    return [pt[idx] for pt in _compute_points(cfg, candles)]


def sweep_warmup(cfg: SweepConfig, output: str) -> int:
    """ATR (read at i-1), the volume window (read before i), and a level's
    whole life plus its pivot. Previous-day levels add a calendar dependency
    no bar count bounds, as prev_hl_warmup notes. 0 for an output this config
    does not expose."""
    if output not in _OUTPUT_INDEX:
        return 0
    return max(
        cfg.atr_len + 1,
        cfg.vol_len + 1 if cfg.use_volume else 0,
        cfg.level_max_age + cfg.pivot_left + cfg.pivot_right,
    )
```

// The SWEEP pane's OUTPUT SHAPE, config parsing and warm-up: a leaf with no
// RUNTIME imports, so exprInstances.ts stays node-safe (same split, same
// reason, as ./spikeOutputs).
//
// Mirrors Python indicators/sweep.py (`parse_sweep_config` / `sweep_outputs`
// / `sweep_warmup`), which is what the backend validates a rule ref against.

/** Rule-operand names in pane order: the SAME strings as the backend's
 * SWEEP_OUTPUTS and the chart figure keys. Fixed names, so retuning a pane
 * keeps every rule that reads it valid. */
export const SWEEP_OUTPUTS = [
  "bearSweep",
  "bullSweep",
  "bearConfirmed",
  "bullConfirmed",
  "score",
  "sweptLevel",
  "barsSince",
] as const;

export type SweepOutput = (typeof SWEEP_OUTPUTS)[number];

export interface SweepConfig {
  pivotLeft: number;
  pivotRight: number;
  levelMaxAge: number; // bars a level stays live after its birth bar
  minWickPct: number; // swept-side wick, % of range, for full wick credit
  atrLen: number;
  minRangeAtr: number; // bar range in ATRs for full range credit
  volLen: number;
  volZ: number;
  minScore: number;
  confirmBars: number;
  usePrevDay: boolean;
  useVolume: boolean;
  // Raw pane zone; computeSweep resolves "", "chart" or invalid to the chart zone.
  tz?: string;
}

export const SWEEP_DEFAULTS: SweepConfig = {
  pivotLeft: 5,
  pivotRight: 5,
  levelMaxAge: 150,
  minWickPct: 50,
  atrLen: 14,
  minRangeAtr: 1.5,
  volLen: 20,
  volZ: 2,
  minScore: 70,
  confirmBars: 3,
  usePrevDay: true,
  useVolume: false,
};

/** calcParams order: [pivotLeft, pivotRight, levelMaxAge, minWickPct,
 * atrLen, minRangeAtr, volLen, volZ, minScore, confirmBars]. Lengths fall
 * back on anything non-finite or 0, positive reals on anything non-finite or
 * <= 0. Mirrored by backend sweep.parse_sweep_config; keep in sync. */
export function parseSweepConfig(calcParams: unknown, extendData?: unknown): SweepConfig {
  const p = Array.isArray(calcParams) ? calcParams : [];
  const ext = (extendData && typeof extendData === "object" ? extendData : {}) as Record<string, unknown>;
  const d = SWEEP_DEFAULTS;
  const lenAt = (i: number, def: number): number => {
    const v = Number(p[i]);
    return Math.max(1, Math.floor(Number.isFinite(v) && v !== 0 ? v : def));
  };
  const posAt = (i: number, def: number): number => {
    const v = Number(p[i]);
    return Number.isFinite(v) && v > 0 ? v : def;
  };
  return {
    pivotLeft: lenAt(0, d.pivotLeft),
    pivotRight: lenAt(1, d.pivotRight),
    levelMaxAge: lenAt(2, d.levelMaxAge),
    minWickPct: posAt(3, d.minWickPct),
    atrLen: lenAt(4, d.atrLen),
    minRangeAtr: posAt(5, d.minRangeAtr),
    volLen: lenAt(6, d.volLen),
    volZ: posAt(7, d.volZ),
    minScore: posAt(8, d.minScore),
    confirmBars: lenAt(9, d.confirmBars),
    usePrevDay: ext.usePrevDay !== false,
    useVolume: ext.useVolume === true,
    tz: typeof ext.tz === "string" ? ext.tz : undefined,
  };
}

/** ATR (read at i-1), the volume window (read before i), and a level's whole
 * life plus its pivot. Every output shares it. Mirrors backend sweep_warmup. */
export function sweepWarmup(cfg: SweepConfig): number {
  return Math.max(
    cfg.atrLen + 1,
    cfg.useVolume ? cfg.volLen + 1 : 0,
    cfg.levelMaxAge + cfg.pivotLeft + cfg.pivotRight,
  );
}

// SWEEP (stop hunt) compute: pivot and previous-day levels, same-bar sweep,
// score and confirmation. Node-safe (type-only klinecharts import) so the
// golden generator and expr layer can run it without a window.
//
// Ported operation-for-operation to backend indicators/sweep.py: keep the
// arithmetic order identical, per the parity contract in indicators/core.py.
// The rules themselves are documented there, once.
import type { KLineData } from "klinecharts";
import { atrSeries } from "../atr";
import { getIndicatorTimezone } from "../indicatorTimezone";
import type { SweepConfig } from "./sweepOutputs";

export interface SweepPoint {
  bearSweep: number;
  bullSweep: number;
  bearConfirmed: number;
  bullConfirmed: number;
  score?: number;
  sweptLevel?: number;
  barsSince?: number;
  // Flag bars only, for the draw: where the swept level's dashed segment
  // starts, and what kind of level it was. Not rule operands.
  levelFrom?: number;
  src?: "pivot" | "day";
}

interface Level {
  price: number;
  born: number;
  start: number;
  src: "pivot" | "day";
}

const DAY_MS = 86_400_000;

const dayFormatters = new Map<string, Intl.DateTimeFormat>();
function dayKey(ts: number, zone: string): number {
  let fmt = dayFormatters.get(zone);
  if (!fmt) {
    fmt = new Intl.DateTimeFormat("en-US", { timeZone: zone, year: "numeric", month: "2-digit", day: "2-digit" });
    dayFormatters.set(zone, fmt);
  }
  let y = 0;
  let m = 0;
  let d = 0;
  for (const p of fmt.formatToParts(ts)) {
    if (p.type === "year") y = Number(p.value);
    else if (p.type === "month") m = Number(p.value);
    else if (p.type === "day") d = Number(p.value);
  }
  return y * 10000 + m * 100 + d;
}

// "", "chart", absent or invalid means the chart zone (PREV_HL's rule). The
// backend gets a stamped zone from withChartTimezone, so both sides agree.
function resolveZone(tz: string | undefined): string {
  if (!tz || tz === "chart") return getIndicatorTimezone();
  try {
    new Intl.DateTimeFormat("en-US", { timeZone: tz });
    return tz;
  } catch {
    return getIndicatorTimezone();
  }
}

// prevHl.ts estimateBarMs without its cache: the median positive gap.
function medianBarMs(dataList: KLineData[]): number {
  if (dataList.length < 2) return 0;
  const deltas: number[] = [];
  for (let i = 1; i < dataList.length; i++) {
    const d = dataList[i].timestamp - dataList[i - 1].timestamp;
    if (d > 0) deltas.push(d);
  }
  return deltas.length ? deltas.sort((a, b) => a - b)[deltas.length >> 1] : 0;
}

const clamp01 = (x: number): number => (x < 0 ? 0 : x > 1 ? 1 : x);

/** Weighted mean of the clamped components, times 100. Range 0 scores 0; a
 * missing ATR or volume reading gives that component 0. */
export function sweepScore(
  cfg: SweepConfig,
  bear: boolean,
  o: number,
  h: number,
  l: number,
  c: number,
  atrPrev: number | null,
  volZ: number | null,
): number {
  const rng = h - l;
  if (rng <= 0) return 0;
  let wick: number;
  let closeD: number;
  if (bear) {
    wick = (h - Math.max(o, c)) / rng;
    closeD = (h - c) / rng;
  } else {
    wick = (Math.min(o, c) - l) / rng;
    closeD = (c - l) / rng;
  }
  const w = clamp01(wick / (cfg.minWickPct / 100));
  const cl = clamp01(closeD / (2 / 3));
  const r = atrPrev !== null && atrPrev > 0 ? clamp01(rng / (cfg.minRangeAtr * atrPrev)) : 0;
  if (cfg.useVolume) {
    const v = volZ !== null ? clamp01(volZ / cfg.volZ) : 0;
    return (w * 0.3 + cl * 0.15 + r * 0.3 + v * 0.25) * 100;
  }
  return (w * 0.4 + cl * 0.2 + r * 0.4) * 100;
}

function volZAt(vols: number[], i: number, n: number): number | null {
  if (i < n) return null;
  let s = 0;
  for (let j = i - n; j < i; j++) s += vols[j];
  const mean = s / n;
  let ss = 0;
  for (let j = i - n; j < i; j++) {
    const d = vols[j] - mean;
    ss += d * d;
  }
  const sd = Math.sqrt(ss / n);
  if (sd === 0) return null;
  return (vols[i] - mean) / sd;
}

function isPivot(vals: number[], p: number, left: number, right: number, wantHigh: boolean): boolean {
  const v = vals[p];
  for (let j = p - left; j <= p + right; j++) {
    if (j === p) continue;
    const w = vals[j];
    if (wantHigh) {
      if (w >= v) return false;
    } else if (w <= v) {
      return false;
    }
  }
  return true;
}

export function computeSweep(dataList: KLineData[], cfg: SweepConfig): SweepPoint[] {
  const n = dataList.length;
  const opens = dataList.map((d) => d.open);
  const highs = dataList.map((d) => d.high);
  const lows = dataList.map((d) => d.low);
  const closes = dataList.map((d) => d.close);
  const vols = dataList.map((d) => d.volume ?? 0);
  const atr = atrSeries(dataList, cfg.atrLen);
  const zone = cfg.usePrevDay && medianBarMs(dataList) <= DAY_MS ? resolveZone(cfg.tz) : null;

  let hiLevels: Level[] = [];
  let loLevels: Level[] = [];
  let curDay: number | null = null;
  let dayHi = -Infinity;
  let dayLo = Infinity;
  let pendDir = 0;
  let pendLevel = 0;
  let pendBar = 0;
  let lastScore: number | undefined;
  let lastLevel: number | undefined;
  let lastBar = -1;

  const out: SweepPoint[] = new Array(n);
  for (let i = 0; i < n; i++) {
    // Births: the pivot at p confirms on this bar.
    const p = i - cfg.pivotRight;
    if (p - cfg.pivotLeft >= 0) {
      if (isPivot(highs, p, cfg.pivotLeft, cfg.pivotRight, true)) hiLevels.push({ price: highs[p], born: i, start: p, src: "pivot" });
      if (isPivot(lows, p, cfg.pivotLeft, cfg.pivotRight, false)) loLevels.push({ price: lows[p], born: i, start: p, src: "pivot" });
    }
    if (zone !== null) {
      const key = dayKey(dataList[i].timestamp, zone);
      if (curDay === null) {
        curDay = key;
      } else if (key !== curDay) {
        hiLevels.push({ price: dayHi, born: i, start: i, src: "day" });
        loLevels.push({ price: dayLo, born: i, start: i, src: "day" });
        curDay = key;
        dayHi = -Infinity;
        dayLo = Infinity;
      }
    }

    // Expiry.
    hiLevels = hiLevels.filter((lv) => i - lv.born <= cfg.levelMaxAge);
    loLevels = loLevels.filter((lv) => i - lv.born <= cfg.levelMaxAge);

    // Crossings: every crossed level dies; the most extreme is the candidate.
    let bearLv: Level | null = null;
    let kept: Level[] = [];
    for (const lv of hiLevels) {
      if (highs[i] > lv.price) {
        if (bearLv === null || lv.price > bearLv.price) bearLv = lv;
      } else {
        kept.push(lv);
      }
    }
    hiLevels = kept;
    let bullLv: Level | null = null;
    kept = [];
    for (const lv of loLevels) {
      if (lows[i] < lv.price) {
        if (bullLv === null || lv.price < bullLv.price) bullLv = lv;
      } else {
        kept.push(lv);
      }
    }
    loLevels = kept;

    const atrPrev = i >= 1 ? atr[i - 1] : null;
    const vz = cfg.useVolume && (bearLv || bullLv) ? volZAt(vols, i, cfg.volLen) : null;
    let bearS: number | null = null;
    let bullS: number | null = null;
    if (bearLv !== null && closes[i] < bearLv.price) {
      const s = sweepScore(cfg, true, opens[i], highs[i], lows[i], closes[i], atrPrev, vz);
      if (s >= cfg.minScore) bearS = s;
    }
    if (bullLv !== null && closes[i] > bullLv.price) {
      const s = sweepScore(cfg, false, opens[i], highs[i], lows[i], closes[i], atrPrev, vz);
      if (s >= cfg.minScore) bullS = s;
    }
    let flag = 0;
    if (bearS !== null && bullS !== null) flag = bearS > bullS ? -1 : bullS > bearS ? 1 : 0;
    else if (bearS !== null) flag = -1;
    else if (bullS !== null) flag = 1;

    // Confirm check of the sweep pending from an earlier bar.
    let bearConf = 0;
    let bullConf = 0;
    if (pendDir !== 0) {
      if (pendDir < 0 && closes[i] > pendLevel) {
        pendDir = 0;
      } else if (pendDir > 0 && closes[i] < pendLevel) {
        pendDir = 0;
      } else if (i - pendBar >= cfg.confirmBars) {
        if (pendDir < 0) bearConf = 1;
        else bullConf = 1;
        pendDir = 0;
      }
    }

    // This bar's flag: it replaces any pending sweep.
    let flagLv: Level | null = null;
    if (flag !== 0) {
      flagLv = flag < 0 ? bearLv : bullLv;
      pendDir = flag;
      pendLevel = flagLv!.price;
      pendBar = i;
      lastScore = (flag < 0 ? bearS : bullS)!;
      lastLevel = flagLv!.price;
      lastBar = i;
    }

    if (zone !== null) {
      dayHi = Math.max(dayHi, highs[i]);
      dayLo = Math.min(dayLo, lows[i]);
    }

    const pt: SweepPoint = {
      bearSweep: flag < 0 ? 1 : 0,
      bullSweep: flag > 0 ? 1 : 0,
      bearConfirmed: bearConf,
      bullConfirmed: bullConf,
    };
    if (lastBar >= 0) {
      pt.score = lastScore;
      pt.sweptLevel = lastLevel;
      pt.barsSince = i - lastBar;
    }
    if (flagLv) {
      pt.levelFrom = flagLv.start;
      pt.src = flagLv.src;
    }
    out[i] = pt;
  }
  return out;
}

```ts
// Sweep (stop hunt): flags candles that trade through a pivot or
// previous-day level and close back on the near side. The math lives in
// ./sweepCompute (node-safe, Python-mirrored); this module only registers
// the template and draws: a tinted box over each flagged candle, a dashed
// segment along the swept level, a tick once the sweep confirms, and a small
// score chip when bars are wide enough to read it.
import {
  type Indicator,
  type IndicatorDrawParams,
  type IndicatorTemplate,
  type KLineData,
} from "klinecharts";
import { DRAW_CLIP_PAD, clipSegmentToRect } from "./shared";
import { computeSweep, type SweepPoint } from "./sweepCompute";
import { parseSweepConfig } from "./sweepOutputs";

export interface SweepExtend {
  hideLegendValue?: boolean;
  // Off-switch for the "72 pivot" chips over flagged candles. Absent means shown.
  showLabels?: boolean;
  usePrevDay?: boolean;
  useVolume?: boolean;
  tz?: string;
}

// Only the price-scaled output is a figure: klinecharts autoscales the
// candle pane over every overlay figure, so 0/1 events or bar counts there
// would crush the price axis (SPIKE's rule). The draw callback paints the
// boxes and returns true, so this figure feeds the legend only.
const SWEEP_FIGURES = [{ key: "sweptLevel", title: "Swept: ", type: "line" }];

const BEAR = "#EF5350";
const BULL = "#26A69A";

function hexWithAlpha(hex: string, alpha: number): string {
  return `${hex}${Math.round(alpha * 255).toString(16).padStart(2, "0")}`;
}

function drawSweep(params: IndicatorDrawParams<SweepPoint, unknown, unknown>): boolean {
  const { ctx, chart, indicator, xAxis, yAxis, bounding } = params;
  const result = (indicator.result ?? []) as SweepPoint[];
  if (!result.length) return true;
  const cfg = parseSweepConfig(indicator.calcParams, indicator.extendData);
  const ext = (indicator.extendData ?? {}) as SweepExtend;
  const dataList = chart.getDataList();
  const barPxRaw = xAxis.convertToPixel(1) - xAxis.convertToPixel(0);
  const half = Number.isFinite(barPxRaw) ? Math.abs(barPxRaw) / 2 : 0;
  const labels = ext.showLabels !== false && half * 2 >= 5;
  // A flag past the right edge can still own a level segment that crosses
  // the pane, so the loop reaches one level lifetime past the visible range.
  const vr = chart.getVisibleRange();
  const lo = Math.max(0, vr.from - 1);
  const hi = Math.min(result.length, vr.to + 1 + cfg.levelMaxAge + cfg.pivotRight);
  const xMin = -DRAW_CLIP_PAD;
  const xMax = bounding.width + DRAW_CLIP_PAD;
  const yMin = -DRAW_CLIP_PAD;
  const yMax = bounding.height + DRAW_CLIP_PAD;

  ctx.save();
  for (let i = lo; i < hi; i++) {
    const p = result[i];
    const bear = p.bearSweep === 1;
    if (!bear && p.bullSweep !== 1) continue;
    const k: KLineData | undefined = dataList[i];
    if (!k || p.sweptLevel === undefined) continue;
    const color = bear ? BEAR : BULL;
    const x = xAxis.convertToPixel(i);
    const yLevel = yAxis.convertToPixel(p.sweptLevel);
    // Level segment, from the level's start bar to the sweep bar.
    const seg = clipSegmentToRect(
      xAxis.convertToPixel(p.levelFrom ?? i), yLevel, x, yLevel, xMin, yMin, xMax, yMax,
    );
    if (seg) {
      ctx.strokeStyle = hexWithAlpha(color, 0.8);
      ctx.lineWidth = 1;
      ctx.setLineDash([4, 3]);
      ctx.beginPath();
      ctx.moveTo(seg[0], seg[1]);
      ctx.lineTo(seg[2], seg[3]);
      ctx.stroke();
      ctx.setLineDash([]);
    }
    if (i > vr.to) continue; // only the segment can reach into the pane
    const yHigh = yAxis.convertToPixel(k.high);
    const yLow = yAxis.convertToPixel(k.low);
    const w = Math.max(3, half * 2);
    ctx.fillStyle = hexWithAlpha(color, 0.12 + 0.18 * ((p.score ?? 0) / 100));
    ctx.fillRect(x - w / 2, yHigh - 2, w, Math.max(1, yLow - yHigh) + 4);
    // Confirm tick, on the sweep candle, when THIS sweep confirmed.
    const c = result[i + cfg.confirmBars];
    const confirmed = c && c.barsSince === cfg.confirmBars && (bear ? c.bearConfirmed === 1 : c.bullConfirmed === 1);
    if (confirmed) {
      const yTick = bear ? yHigh - 8 : yLow + 8;
      ctx.strokeStyle = color;
      ctx.lineWidth = 1.5;
      ctx.beginPath();
      ctx.moveTo(x - 3, yTick);
      ctx.lineTo(x - 1, yTick + 2);
      ctx.lineTo(x + 3, yTick - 2);
      ctx.stroke();
    }
    if (labels) {
      const text = `${Math.round(p.score ?? 0)} ${p.src === "day" ? "prev day" : "pivot"}`;
      ctx.font = "600 10px ui-monospace, Menlo, monospace";
      const tw = ctx.measureText(text).width + 8;
      const yBase = bear ? yHigh - 14 : yLow + 24;
      ctx.fillStyle = hexWithAlpha(color, 0.92);
      ctx.beginPath();
      ctx.roundRect(x - tw / 2, yBase - 11, tw, 14, 3);
      ctx.fill();
      ctx.fillStyle = "#FFFFFF";
      ctx.textAlign = "left";
      ctx.textBaseline = "alphabetic";
      ctx.fillText(text, x - tw / 2 + 4, yBase);
    }
  }
  ctx.restore();
  return true; // boxes replace the default figure lines
}

// calcParams = [pivotLeft, pivotRight, levelMaxAge, minWickPct, atrLen,
// minRangeAtr, volLen, volZ, minScore, confirmBars].
export const SWEEP_TEMPLATE: Omit<IndicatorTemplate, "name"> = {
  shortName: "Stop Hunt",
  series: "price",
  precision: 2,
  calcParams: [5, 5, 150, 50, 14, 1.5, 20, 2, 70, 3],
  figures: SWEEP_FIGURES,
  calc: (dataList: KLineData[], ind: Indicator) =>
    computeSweep(dataList, parseSweepConfig(ind.calcParams, ind.extendData)),
  draw: (params) => drawSweep(params as IndicatorDrawParams<SweepPoint, unknown, unknown>),
};
```
